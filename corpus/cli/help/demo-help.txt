usage: termproof demo [-h] [--out OUT] [--no-open] [--video]
                      [--video-fps VIDEO_FPS] [--config CONFIG]
                      [--reporter REPORTER] [--xml-path XML_PATH]
                      [--screen-renderer SCREEN_RENDERER]
                      [--video-backend VIDEO_BACKEND]

options:
  -h, --help            show this help message and exit
  --out OUT             output directory for demo evidence (default:
                        .termproof/demo)
  --no-open             do not attempt to open generated report in browser
  --video               render video evidence if video backend available
  --video-fps VIDEO_FPS
  --config CONFIG       path to a termproof config YAML file
  --reporter REPORTER   reporter to use (default: markdown, also: junit_xml)
  --xml-path XML_PATH   additional path to write JUnit XML report
  --screen-renderer SCREEN_RENDERER
                        screen renderer (default: svg)
  --video-backend VIDEO_BACKEND
                        video backend (default: agg_ffmpeg)
